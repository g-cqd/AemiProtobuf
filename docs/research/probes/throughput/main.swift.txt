import SwiftProtobuf
import Foundation

func makeFast(_ n: Int) -> [FUser] {
    (0..<n).map { i in
        FUser(id: Int64(i) * 7919 + 1, name: "user-\(i)-name", email: "user\(i)@example.com", active: i % 2 == 0,
              score: Double(i) * 0.25 + 0.5, tags: ["alpha", "beta\(i % 10)", "gamma"], samples: (0..<8).map { Int32($0 * 37 - 5 + i) },
              address: FAddress(street: "\(i) Main Street", city: "Springfield", zip: Int32(10000 + i % 9000)),
              delta: Int32(i % 50) - 25, stamp: UInt64(1_700_000_000 + i))
    }
}
func toPB(_ f: FUser) -> User {
    var u = User(); u.id = f.id; u.name = f.name; u.email = f.email; u.active = f.active; u.score = f.score
    u.tags = f.tags; u.samples = f.samples; u.delta = f.delta; u.stamp = f.stamp
    if let a = f.address { var x = Address(); x.street = a.street; x.city = a.city; x.zip = a.zip; u.address = x }
    return u
}
func median(_ runs: Int, _ f: () throws -> Void) rethrows -> Double {
    var t: [Double] = []
    for _ in 0..<runs {
        let c = ContinuousClock(); let d = try c.measure { try f() }
        t.append(Double(d.components.attoseconds) / 1e18 + Double(d.components.seconds))
    }
    t.sort(); return t[t.count / 2]
}
